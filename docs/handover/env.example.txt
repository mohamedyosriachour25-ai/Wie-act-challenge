# HerwieGO environment template
# ---------------------------------------------------------------------------
# For a LOCAL (non-Docker) run: copy the relevant lines into  backend/.env
# For DOCKER: copy them into  ./.env  at the repo root (docker-compose reads it)
# NEVER commit a real .env. NEVER paste the real team API key into logs/screenshots.
# The repo already ships two working templates: ./.env.example and backend/.env.example
# ---------------------------------------------------------------------------

# --- AI inference service -------------------------------------------------
# Both required, in every environment -- there is no built-in fallback and the app refuses to
# start without them. Set the base URL + team key + the model ids from GET /v1/models.
HERWIEGO_BASE_URL=
HERWIEGO_API_KEY=
HERWIEGO_COMPACT_MODEL=compact-1b
HERWIEGO_MID_MODEL=mid-8b
HERWIEGO_EMBED_MODEL=embed-small
HERWIEGO_MIN_SIMILARITY=0.1          # relevance floor; retrieval also keeps only passages >= half the best score
HERWIEGO_VECTOR_BACKEND=local        # vectors in the app's own database (supported setup)
HERWIEGO_OCR_MODEL=deepseek-ocr      # reads scanned PDF pages (one call per page); empty = keep scans as files
HERWIEGO_OCR_MAX_PAGES=40
HERWIEGO_MAX_PDF_MB=20
HERWIEGO_MAX_CONCURRENT=2            # per-key concurrency ceiling

# --- Session signing ------------------------------------------------------
# Random secret for HMAC-signed bearer tokens. Generate: openssl rand -hex 32
# PowerShell: -join ((1..32) | % { '{0:x2}' -f (Get-Random -Max 256) })
# Production REFUSES to start if this is weak (<24 chars) or a known placeholder.
HERWIEGO_SECRET=change-me-for-any-shared-deployment

# --- App runtime ----------------------------------------------------------
HERWIEGO_DB=herwiego.db               # SQLite file path (Docker uses /data/herwiego.db)
HERWIEGO_ENV=development            # "production" hides /docs and enforces secret/key
HERWIEGO_LOG_LEVEL=INFO
HERWIEGO_TRUST_PROXY=0              # 1 only behind your own reverse proxy that sets X-Forwarded-For

# --- Rate limits (protect the shared team quota) --------------------------
HERWIEGO_USER_AI_PER_MIN=20         # AI actions one learner may start per minute
HERWIEGO_LOGIN_ATTEMPTS=8           # failed sign-ins per 5 min per pseudo (per-connection ceiling is 5x)
HERWIEGO_REGISTER_PER_HOUR=15

# --- Live rooms -----------------------------------------------------------
HERWIEGO_VIDEO_CAPACITY=9           # cameras sharing one room wall at once (WebRTC mesh)
HERWIEGO_LIVE_CAPACITY=200          # presence/chat capacity of a public room
HERWIEGO_EVENT_LEAD_S=300           # seconds before a break the group-game invite opens
HERWIEGO_EVENT_MIN_PLAYERS=2        # minimum players referenced by the invite

# --- AI optimisation ------------------------------------------------------
HERWIEGO_SEMANTIC_CACHE=1           # reuse results for near-duplicate requests (embeddings)
HERWIEGO_SEM_THRESHOLD=0.86         # cosine above which an explanation is reused
HERWIEGO_SEM_THRESHOLD_ASK=0.93     # stricter threshold for material questions
HERWIEGO_QUIZ_VERIFY=1              # draft-then-revise: compact pass reviews the quiz before display
HERWIEGO_MAX_QUOTA_WAIT_S=45        # wait at most this long on quota before telling the user

# Mandatory in every environment: HERWIEGO_SECRET (strong in production), HERWIEGO_BASE_URL, HERWIEGO_API_KEY,
#   HERWIEGO_COMPACT_MODEL, HERWIEGO_MID_MODEL, HERWIEGO_EMBED_MODEL.
# Everything else is optional and has a sensible default.
