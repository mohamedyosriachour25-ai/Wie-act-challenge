# EdNova environment template
# ---------------------------------------------------------------------------
# For a LOCAL (non-Docker) run: copy the relevant lines into  backend/.env
# For DOCKER: copy them into  ./.env  at the repo root (docker-compose reads it)
# NEVER commit a real .env. NEVER paste the real team API key into logs/screenshots.
# The repo already ships two working templates: ./.env.example and backend/.env.example
# ---------------------------------------------------------------------------

# --- AI inference service -------------------------------------------------
# "mock" runs the built-in fake service in-process (no key needed; canned lessons).
# For the real challenge service, set the base URL + team key + the model ids from GET /v1/models.
EDNOVA_BASE_URL=mock
EDNOVA_API_KEY=replace-me
EDNOVA_COMPACT_MODEL=compact-1b
EDNOVA_MID_MODEL=mid-8b
EDNOVA_EMBED_MODEL=embed-small
EDNOVA_MIN_SIMILARITY=0.1          # cosine below this = passage not relevant (tune on real embeddings)
EDNOVA_MAX_CONCURRENT=2            # per-key concurrency ceiling

# --- Session signing ------------------------------------------------------
# Random secret for HMAC-signed bearer tokens. Generate: openssl rand -hex 32
# PowerShell: -join ((1..32) | % { '{0:x2}' -f (Get-Random -Max 256) })
# Production REFUSES to start if this is weak (<24 chars) or a known placeholder.
EDNOVA_SECRET=change-me-for-any-shared-deployment

# --- App runtime ----------------------------------------------------------
EDNOVA_DB=ednova.db               # SQLite file path (Docker uses /data/ednova.db)
EDNOVA_ENV=development            # "production" hides /docs and enforces secret/key
EDNOVA_LOG_LEVEL=INFO
EDNOVA_TRUST_PROXY=0              # 1 only behind your own reverse proxy that sets X-Forwarded-For

# --- Rate limits (protect the shared team quota) --------------------------
EDNOVA_USER_AI_PER_MIN=20         # AI actions one learner may start per minute
EDNOVA_LOGIN_ATTEMPTS=8           # failed sign-ins per 5 min per pseudo (per-connection ceiling is 5x)
EDNOVA_REGISTER_PER_HOUR=15

# --- Live rooms -----------------------------------------------------------
EDNOVA_VIDEO_CAPACITY=9           # cameras sharing one room wall at once (WebRTC mesh)
EDNOVA_LIVE_CAPACITY=200          # presence/chat capacity of a public room
EDNOVA_EVENT_LEAD_S=300           # seconds before a break the group-game invite opens
EDNOVA_EVENT_MIN_PLAYERS=2        # minimum players referenced by the invite

# --- AI optimisation ------------------------------------------------------
EDNOVA_SEMANTIC_CACHE=1           # reuse results for near-duplicate requests (embeddings)
EDNOVA_SEM_THRESHOLD=0.86         # cosine above which an explanation is reused
EDNOVA_SEM_THRESHOLD_ASK=0.93     # stricter threshold for material questions
EDNOVA_QUIZ_VERIFY=1              # draft-then-revise: compact pass reviews the quiz before display
EDNOVA_MAX_QUOTA_WAIT_S=45        # wait at most this long on quota before telling the user
EDNOVA_MOCK_LATENCY_SCALE=1       # 0 = instant mock replies (used by the test suite)
